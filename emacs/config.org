  (defun my/vertico-kill-buffer ()
    "Kill buffer candidate at point and remove from Vertico list."
    (interactive)
    (when-let ((cand (vertico--candidate)))
      (let* ((multi (get-text-property 0 'multi-category cand))
              (target (if (and multi (eq (car multi) 'buffer))
                        (cdr multi)
                        cand))
              ;; Target can be a buffer object or string candidate
              (buf (if (bufferp target)
                     target
                     (get-buffer (replace-regexp-in-string "[\x100000-\x10fffd]" ""
                                   (substring-no-properties target)))))
              (name (and (buffer-live-p buf) (buffer-name buf))))
        (if (not (buffer-live-p buf))
          (message "Candidate is not a live buffer: %s" cand)
          ;; Prompt if modified file buffer, else kill
          (if (and (buffer-modified-p buf) (buffer-file-name buf))
            (when (yes-or-no-p (format "Buffer %s modified. Kill anyway? " name))
              (kill-buffer buf))
            (kill-buffer buf))
          ;; Remove from Vertico view immediately
          (setq vertico--candidates (delete cand vertico--candidates))
          (setq vertico--total (max 0 (1- vertico--total)))
          (vertico--exhibit)
          (message "Killed buffer: %s" name)))))

  (defvar my/compile-timings (make-hash-table :test 'equal))
  (defvar my/compile-current-perf-file nil)

  ;; 2. Project marker table (Outermost root match)
  (defvar my/project-build-tools
    '(("build.sh"       . (lambda (root _dir _file _base)
                            (format "./build.sh")))
       ("Makefile"       . (lambda (root _dir _file _base)
                             (format "make -B")))
       ("CMakeLists.txt" . (lambda (root _dir _file _base)
                             (format "mkdir -p build && cd build && cmake .. && cmake --build . -j $(nproc)")))
       ("nob.c"          . (lambda (root _dir _file _base)
                             (let ((src (shell-quote-argument (concat root "/nob.c")))
                                    (bin (shell-quote-argument (concat root "/nob"))))
                               (format "_perf 'compilation' && cc -Wall -Wextra -o %s %s && _perf 'run' && %s"
                                 bin src bin))))
       ("Cargo.toml"     . (lambda (_root _dir _file _base)
                             "cargo run --color=always"))
       ("*.csproj"       . (lambda (_root _dir _file _base)
                             "dotnet run"))))

  ;; 3. Single-file runner table (Grouped by major-mode)
  (defvar my/file-runners
    '(((c-mode c-ts-mode) .
        (lambda (dir file base)
          (format "_perf 'compilation' && cd %s && mkdir -p ./bin && cc -Wall -Wextra -ggdb -fdiagnostics-color=always -o ./bin/%s %s && _perf 'run' && ./bin/%s"
            dir base file base)))

       ((c++-mode c++-ts-mode) .
         (lambda (dir file base)
           (format "_perf 'compilation' && cd %s && mkdir -p ./bin && g++ -Wall -Wextra -ggdb -fdiagnostics-color=always -o ./bin/%s %s && _perf 'run' && ./bin/%s"
             dir base file base)))

       ((csharp-mode csharp-ts-mode) .
         (lambda (dir file base)
           (format "_perf 'compilation' && dotnet run --project %s || (csc %s && _perf 'run' && mono %s.exe)"
             dir file base)))

       ((python-mode python-ts-mode) .
         (lambda (_dir file _base)
           (format "_perf 'run' && python3 %s" file)))

       ((php-mode) .
         (lambda (_dir file _base)
           (format "_perf 'run' && php %s" file)))

       ((lua-mode) .
         (lambda (_dir file _base)
           (format "_perf 'run' && lua %s" file)))

       ((rust-mode rust-ts-mode) .
         (lambda (dir file base)
           (format "_perf 'compilation' && rustc --color=always %s -o %s/%s && _perf 'run' && %s/%s"
             file dir base dir base)))

       ((java-mode java-ts-mode) .
         (lambda (dir file base)
           (let ((awk "awk '{ gsub(/[^ \\t:]+\\.java/, \"\\033[1;33m&\\033[0m\"); gsub(/errors|error:/, \"\\033[1;31m&\\033[0m\"); gsub(/warning:/, \"\\033[1;33mwarning:\\033[0m\"); print }'"))

  ;; 4. Find outermost project root
  (defun my/find-project-root (start-dir)
    "Walk up to '/' to locate outermost folder with build tool marker."
    (let ((current (expand-file-name start-dir))
           (outermost-root nil)
           (outermost-marker nil))
      (while (not (equal current "/"))
        (dolist (tool my/project-build-tools)
          (let ((marker (car tool)))
            (if (string-equal marker "*.csproj")
              (when (directory-files current nil "\\.csproj$" t)
                (setq outermost-root current
                  outermost-marker marker))
              (when (file-exists-p (expand-file-name marker current))
                (setq outermost-root current
                  outermost-marker marker)))))
        (setq current (file-name-directory (directory-file-name current))))
      (cons outermost-root outermost-marker)))

  ;; 5. Main compile command
  (defun my/auto-compile ()
    "Save all buffers, resolve command via root marker or major-mode, run."
    (interactive)
    (save-some-buffers t) ; vim.cmd("wall") equivalent
    (clrhash my/compile-timings)

    (let* ((file (buffer-file-name))
            (dir (and file (file-name-directory file)))
            (base (and file (file-name-sans-extension (file-name-nondirectory file))))
            (root-info (and dir (my/find-project-root dir)))
            (root (car root-info))
            (marker (cdr root-info))
            (exec-name (or base "compile"))
            (perf-script (format "/tmp/%s" exec-name))
            (perf-file (concat perf-script ".perf"))
            (cwd dir)
            (raw-cmd nil))

      (unless file
        (user-error "Buffer is not visiting a file"))

      (setq my/compile-current-perf-file perf-file)
      (when (file-exists-p perf-file)
        (delete-file perf-file))

      ;; Resolve command
      (if (and root marker)
        (progn
          (setq cwd root)
          (let ((handler (cdr (assoc marker my/project-build-tools))))
            (when handler
              (setq raw-cmd (funcall handler
                              (shell-quote-argument root)
                              (shell-quote-argument dir)
                              (shell-quote-argument file)
                              (shell-quote-argument base))))))
        ;; Fallback to file runners
        (let ((handler (cl-some (lambda (entry)
                                  (when (memq major-mode (car entry))
                                    (cdr entry)))
                         my/file-runners)))
          (when handler
            (setq raw-cmd (funcall handler
                            (shell-quote-argument dir)
                            (shell-quote-argument (file-name-nondirectory file))
                            (shell-quote-argument base))))))

      (unless raw-cmd
        (user-error "No build runner for mode %s" major-mode))

      ;; Inject bash timing harness
      (with-temp-file perf-script
        (insert (format "_perf() {\n    local _n=$(date +%%s%%N)\n    printf \"[%%sms] %%s\\n\" \"$(( (_n - _t) / 1000000 ))\" \"$*\" >> %s\n    _t=$_n\n}\n_t=$(date +%%s%%N)\n" perf-file)))

      (let* ((full-cmd (format ". %s && rm %s && %s"
                         perf-script perf-script raw-cmd))
              (default-directory cwd)
              (compilation-environment '("FORCE_COLOR=1" "CLICOLOR_FORCE=1" "PYTHONUNBUFFERED=1")))
        ;; Run in bottom split window
        (compile full-cmd))))

  ;; 6. Process timing on finish & handle ANSI colors

    "Parse perf file when compilation finishes."

        (goto-char (point-min))
        (while (re-search-forward "\\[\\([0-9]+\\)ms\\][ \t]+\\(.+\\)" nil t)
          (puthash (match-string 2) (concat (match-string 1) "ms") my/compile-timings)))
      (delete-file my/compile-current-perf-file))
    (force-mode-line-update t))

  (global-display-fill-column-indicator-mode 1)
  (setq-default display-fill-column-indicator-column 80)
  ;; Smooth scrolling. Defaults are scroll-step 0 / scroll-conservatively 0,
  ;; which mean "always RECENTER point when it moves off-screen" -> that's the
  ;; snap-to-middle jump at top/bottom, and every recenter is a whole-window
  ;; redraw (a big trigger for ghost-text debris under XWayland on fast key
  ;; movement). scroll-conservatively > 100 makes redisplay NEVER recenter and
  ;; instead scroll just enough to bring point into view, even from far away.
  (setq scroll-step 1)
  (setq scroll-conservatively 101)
  (setq scroll-margin 9)
